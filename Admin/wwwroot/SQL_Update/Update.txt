ALTER PROCEDURE [dbo].[ReOrderStory]
AS
BEGIN
    SET NOCOUNT ON;

    ;WITH CTE AS
    (
        SELECT
            Id,
            NewReadOrder = ROW_NUMBER() OVER (ORDER BY ReadOrder, Id)
        FROM Story
        WHERE ReadOrder IS NOT NULL
          AND ReadOrder <> 0
    )
    UPDATE S
    SET ReadOrder = CTE.NewReadOrder
    FROM Story S
    INNER JOIN CTE
        ON S.Id = CTE.Id;
END

ALTER PROCEDURE [dbo].[Read_Change]
	@id int,
	@idChapter bigint
as
begin
	declare @read bit = (select IsRead from Story where Id = @id)
	Update Story 
	set IsRead = case when @read = 1 then 0 
					when @read = 0 or @read is null then 1 end,
		ReadTo = case when @read = 1 then null
					when @read = 0 or @read is null then @idChapter end,
		ReadOrder = case when @read = 0 then null else ReadOrder end
	where Id =@id
end

create  PROCEDURE [dbo].[CreateOrUpdate_ChapterPatch]
	@Id int,
	@chapterId bigint,
	@verName nvarchar(150),
	@title nvarchar(1000),
	@patch nvarchar(max)
as
begin
	if @Id <= 0
	begin
		if @verName = '' or @verName is null
		begin
			declare @numPatch tinyint = (select count(Id) from ChapterPatch where ChapterId = @chapterId)
			set @verName = 'Version ' + CAST(@numPatch + 1 AS NVARCHAR(10))
		end
		insert into ChapterPatch
		values (@chapterId, @verName, @title, @patch)
	end
	else
		update ChapterPatch
		set Title = @title, Patch = @patch
		where Id = @Id
end

create PROCEDURE [dbo].[Delete_ChapterPatch]
	@id int
as
begin
    DECLARE @ChapterId BIGINT;

    SELECT @ChapterId = ChapterId
    FROM ChapterPatch
    WHERE Id = @Id;

	delete ChapterPatch where Id = @id

    declare @data table (
		Id int,
        Orders int
	);

    insert into @data
    select Id, ROW_NUMBER() OVER (
            ORDER BY CAST(REPLACE(VerName, 'Version ', '') AS INT)
        )
    from ChapterPatch
    where ChapterId = @ChapterId

    update ChapterPatch
    set VerName = 'Version ' + CAST(d.Orders AS NVARCHAR(10))
    from ChapterPatch join @data d on ChapterPatch.Id = d.Id
end
